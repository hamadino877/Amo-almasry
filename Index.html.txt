      // Helper: filter categories based on activeTab & search (unless printing override)
      function getFilteredMenu() {
        if (isPrintingMode) {
          // printing: show everything, ignore tabs & search
          return menuData.map(cat => ({
            ...cat
          }));
        }
        let filtered = [...menuData];
        // filter by category tab
        if (activeTab !== 'all') {
          filtered = filtered.filter(cat => cat.id === activeTab);
        }
        // filter by search query (case-insensitive)
        if (searchQuery.trim() !== '') {
          const q = searchQuery.toLowerCase();
          filtered = filtered.map(category => {
            // filter main items
            const matchedItems = category.items.filter(item => item.name.toLowerCase().includes(q));
            let matchedSub = null;
            if (category.subSection) {
              const matchedSubItems = category.subSection.items.filter(subItem => subItem.name.toLowerCase().includes(q));
              if (matchedSubItems.length > 0) {
                matchedSub = {
                  ...category.subSection,
                  items: matchedSubItems
                };
              }
            }
            if (matchedItems.length > 0 || matchedSub) {
              return {
                ...category,
                items: matchedItems,
                subSection: matchedSub || null
              };
            }
            return null;
          }).filter(cat => cat !== null);
        }
        return filtered;
      }

										<h3 class="font-amiri text-xl font-bold text-[#6B4F0E] mb-4 flex items-center gap-2">`;
            html += `
											<span>${sub.icon || '🍯'}</span> ${sub.title}
										</h3>`;
            html += `
										<div class="grid grid-cols-1 md:grid-cols-2 gap-4 print-grid">`;
            for (const subItem of sub.items) {
              html += `
											<div class="bg-white border border-[#EBE3CD] rounded-xl p-3 font-bold text-sm">${subItem.name}</div>`;
            }
            html += `
										</div>
									</div>`;
          }
          html += `
								</section>`;
        }
        menuContainer.innerHTML = html;
      }
      // build tabs dynamically from menuData
      function buildTabs() {
        let tabsHtml = `
								<button data-tab="all" class="tab-btn whitespace-nowrap px-5 py-2 rounded-full text-sm font-bold transition-all ${activeTab === 'all' ? 'tab-active' : 'tab-inactive'}">الكل</button>`;
        for (const cat of menuData) {
          const isActive = activeTab === cat.id;
          tabsHtml += `
								<button data-tab="${cat.id}" class="tab-btn whitespace-nowrap flex items-center gap-1.5 px-4 py-2 rounded-full text-sm font-bold transition-all ${isActive ? 'tab-active' : 'tab-inactive'}">
									<span>${cat.icon}</span> ${cat.title}
                            
								</button>`;
        }
        tabsContainer.innerHTML = tabsHtml;
        // attach click events to tab buttons
        document.querySelectorAll('.tab-btn').forEach(btn => {
          btn.addEventListener('click', (e) => {
            if (isPrintingMode) return;
            const tabId = btn.getAttribute('data-tab');
            activeTab = tabId;
            // re-render tabs style and menu
            buildTabs(); // refresh active style
            renderMenu();
          });
        });
      }
      // handle search input
      function bindSearch() {
        searchInput.addEventListener('input', (e) => {
          if (isPrintingMode) return;
          searchQuery = e.target.value;
          renderMenu();
          // optionally, keep tabs active style as is, but no need to rebuild tabs
        });
      }
      // ---------- PRINT LOGIC (full override, then restore) ----------
      async function handlePrint() {
        // save current state
        previousState = {
          activeTab: activeTab,
          searchQuery: searchQuery
        };
        // enable printing mode: override filters to show all categories and no search
        isPrintingMode = true;
        // force refresh menu (show all)
        renderMenu();
        // small delay to ensure DOM is updated with full menu for print
        setTimeout(() => {
          window.print();
          // after print dialog, restore previous state
          setTimeout(() => {
            isPrintingMode = false;
            activeTab = previousState.activeTab;
            searchQuery = previousState.searchQuery;
            // update search input value
            searchInput.value = searchQuery;
            // rebuild tabs and render menu with restored filters
            buildTabs();
            renderMenu();
          }, 100);
        }, 200);
      }
      // initial load & event binding
      function init() {
        buildTabs();
        bindSearch();
        renderMenu();
        printBtn.addEventListener('click', handlePrint);
      }
      // start everything
      init();
    </script>
